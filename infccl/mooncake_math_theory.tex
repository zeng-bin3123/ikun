\section{Prefill Cost Model and KVCache Reuse Benefit}
\label{sec:prefill-cost}

\subsection{Foundational FLOP Model (FAST25 Paper, Equation 1)}

\begin{commitbox}{Mooncake FAST25-release --- Mooncake-FAST25.pdf \S2.2, Table~1}
The Mooncake paper establishes the foundational cost model for prefill computation.
Model parameters: $l=80$ layers, $d=8192$ model dimension, $a=4,b=22$ attention/FFN
coefficients, $\text{gqa}=8$ (GQA ratio), $s=2$ bytes (BFloat16).
\end{commitbox}

\begin{definition}[Prefill FLOP Count]
\label{def:prefill-flops}
For a transformer model with $l$ layers, model dimension $d$, and input length $n$:
\begin{equation}
  \mathrm{flops}(n) \;=\; l \cdot \bigl(a\,n^2\,d \;+\; b\,n\,d^2\bigr)
  \label{eq:prefill-flops}
\end{equation}
where $a$ captures the quadratic self-attention cost ($Q K^T$ and $\mathrm{softmax} \cdot V$)
and $b$ captures the linear feed-forward cost.
\end{definition}

\begin{theorem}[KVCache Reuse Benefit Criterion]
\label{thm:kv-reuse}
Given a prefix match of length $p$ out of total input $n$, KVCache transfer bandwidth $B$,
and GPU throughput $G$, reuse is beneficial for TTFT if and only if:
\begin{equation}
  \underbrace{\frac{l \cdot (a\,p^2\,d + b\,p\,d^2)}{G}}_{\text{prefill compute saved}}
  \;>\;
  \underbrace{\frac{p \cdot l \cdot 2d \cdot s / \text{gqa}}{B}}_{\text{KVCache transfer cost}}
  \label{eq:kv-reuse-criterion}
\end{equation}
\end{theorem}

\begin{proof}
The prefill compute saved by reusing $p$ tokens is exactly
$\mathrm{flops}(n) - \mathrm{flops}(n-p)$.  Expanding:
\begin{align*}
  \Delta_{\text{flops}} &= l\bigl[a(n^2 - (n-p)^2)d + b(n-(n-p))d^2\bigr]\\
  &= l\bigl[a(2np - p^2)d + bpd^2\bigr]
\end{align*}
For prefix caching where $p$ is the matched prefix, the dominant term when $p \ll n$ is
approximately $l(ap^2 d + bpd^2)$.  The KVCache size per token is
$l \cdot 2 \cdot (d/\text{gqa}) \cdot s$ bytes (key + value across all layers), so transferring
$p$ tokens costs $p \cdot l \cdot 2d \cdot s / \text{gqa}$ bytes at bandwidth $B$.
The criterion follows from comparing compute time $\Delta_{\text{flops}}/G$ against
transfer time.
\end{proof}

\begin{corollary}[Minimum Bandwidth for BI-V100]
\label{cor:biv100-bandwidth}
For BI-V100 ($G \approx 4 \times 16.3 = 65.2$ TFLOPS, $B_{\text{pcie}} \approx 16$~GB/s
per direction), with a hypothetical 70B-class model ($l=80, d=8192$) and $p=8192$ prefix:
\[
  B_{\min} \;=\; \frac{p \cdot l \cdot 2d \cdot s / \text{gqa} \cdot G}
  {l \cdot (ap^2 d + bpd^2)}
  \;=\; \frac{p \cdot 2ds / \text{gqa}}{ap^2d/G + bpd^2/G}
\]
Since BI-V100 has only PCIe P2P (no NVLink/IB), the inter-GPU bandwidth
$B \leq 16$~GB/s sets a hard constraint on maximum reusable prefix length.
\end{corollary}

\begin{definition}[RDMA Completion Queue Entry (CQE) Model]
\label{def:cqe-model}
An RDMA Send Queue (SQ) of depth $W$ posts Work Requests (WRs).  Each signaled WR
generates one CQE upon completion.  The CQ has finite depth $C$.  Standard practice
signals every WR, giving a 1:1 CQE-to-WR ratio.  We define:
\begin{itemize}[nosep]
  \item $N$: total WRs in a posting chain
  \item $k$: signal period, i.e., one CQE per $k$ WRs
  \item $W$: SQ depth (from \texttt{ibv\_create\_qp} \texttt{cap.max\_send\_wr})
\end{itemize}
\end{definition}

\begin{theorem}[CQE Reduction Factor under Amortized Signaling]
\label{thm:cqe-reduction}
Let $N$ WRs be posted in a chain of length $N$, with signal period
$k = \min(\text{interval}, \lfloor W/2 \rfloor)$.  The number of CQEs generated is:
\begin{equation}
  C_k(N) \;=\; \left\lfloor \frac{N}{k} \right\rfloor \;+\;
  \begin{cases} 1 & \text{if } N \bmod k \neq 0 \\ 0 & \text{otherwise} \end{cases}
  \;\leq\; \left\lceil \frac{N}{k} \right\rceil
  \label{eq:cqe-count}
\end{equation}
The CQE reduction factor is:
\begin{equation}
  R_{\text{CQE}} \;=\; \frac{N}{C_k(N)} \;\approx\; k
  \quad\text{for } N \gg k
  \label{eq:cqe-reduction}
\end{equation}
\end{theorem}

\begin{proof}
The signaling predicate from \texttt{shouldSignalRdmaWr} is:
\[
  \text{signal}(i) \;=\; \bigl[(i+1) \bmod k = 0\bigr] \;\lor\; \bigl[i+1 = N\bigr]
\]
Counting: indices $\{k-1, 2k-1, \ldots, \lfloor N/k\rfloor \cdot k - 1\}$ satisfy the
modular condition, giving $\lfloor N/k \rfloor$ CQEs.  If $N$ is not a multiple of $k$,
the chain tail at index $N-1$ adds one more CQE.  The bound $\lceil N/k \rceil$ follows.

The constraint $k \leq \lfloor W/2 \rfloor$ from the commit's
``\textit{half-SQ signal period as pipeline, not deadlock}'' comment ensures that
the SQ is freed in at most two batches: the first signaled WR retires $k$ slots,
leaving $\geq W/2$ slots for subsequent posts before the next CQE arrives.
\end{proof}

\begin{lemma}[SQ Occupancy Bound]
\label{lem:sq-occupancy}
With signal period $k \leq \lfloor W/2 \rfloor$, the maximum number of in-flight
(unretired) WRs at any point is bounded by $W$, and the pipeline never stalls provided
the CQ polling thread processes each CQE before $k$ new WRs accumulate.
\end{lemma}

\begin{proof}
Each CQE retires exactly $k$ WRs from the FIFO (\texttt{collectPostedCompletions}).
Since the tail of each chain is always signaled, the SQ depth returns to 0 after at most
$\lceil N/k \rceil$ poll cycles.  The SQ capacity $W$ allows $\lfloor W/k \rfloor$
unsignaled bursts before the first CQE must be polled.  Setting $k \leq W/2$ guarantees
$\lfloor W/k \rfloor \geq 2$, so at least two full bursts can be posted before any
CQE is required.
\end{proof}

\begin{lemma}[Flushed WR Error Propagation]
\label{lem:flush-err}
When a QP enters error state, all subsequent unsignaled WRs are flushed with
\texttt{WR\_FLUSH\_ERR}. The posted FIFO marks the prefix before the first error CQE as
\texttt{SUCCESS} (RC in-order guarantee) and the remainder as \texttt{FLUSH\_ERR}:
\begin{equation}
  \text{status}(WR_j) =
  \begin{cases}
    \text{SUCCESS} & j < j_{\text{error}} \\
    \text{ERROR} & j = j_{\text{error}} \\
    \text{FLUSH\_ERR} & j > j_{\text{error}}
  \end{cases}
\end{equation}
This is sound because InfiniBand RC QPs guarantee in-order completion: if WR $j_{\text{error}}$
failed, all WRs $j < j_{\text{error}}$ have already completed successfully on the wire.
\end{lemma}

\begin{remark}[BI-V100 Application]
On BI-V100's PCIe P2P path, the ``RDMA'' equivalent is CUDA IPC memory copies.
The amortized signaling principle maps to \textbf{batched CUDA event synchronization}:
instead of \texttt{cudaEventRecord} + \texttt{cudaEventSynchronize} per P2P copy,
infccl records an event every $k$ operations and synchronizes on the batch.
The \texttt{INFCCL\_BATCH\_BARRIER} environment variable controls this parameter.
\end{remark}

\begin{theorem}[String Interning Complexity Reduction]
\label{thm:string-intern}
Let $S$ be the number of slices submitted per batch, $P$ the number of distinct
peer NIC paths (cardinality $|\mathcal{P}|$), and $\ell$ the average path string length.
Without interning, the per-batch allocation cost is $\Theta(S \cdot \ell)$ bytes with
$\Theta(S)$ allocator calls.  With interning:
\begin{equation}
  \text{Alloc}_{\text{intern}} \;=\;
  \underbrace{\BigO(|\mathcal{P}| \cdot \ell)}_{\text{one-time pool build}} \;+\;
  \underbrace{\BigO(S)}_{\text{pointer copies}}
  \label{eq:intern-cost}
\end{equation}
The per-slice path resolution drops from $\Theta(\ell)$ (string copy + hash) to
$\Theta(1)$ (pointer dereference).
\end{theorem}

\begin{proof}
The intern pool uses an \texttt{unordered\_map} with amortized $\BigO(1)$ lookup.
Since $|\mathcal{P}| = O(\text{peers} \times \text{NICs})$ is bounded by the cluster
topology (typically $\leq 10^3$), the pool is populated in
$\BigO(|\mathcal{P}| \cdot \ell)$ time during \texttt{decodeSegmentDesc}.
Each slice's \texttt{interned\_peer\_nic\_path} is a raw pointer into the pool
(never freed during process lifetime), so assignment and comparison are $\Theta(1)$.

The critical hot-path savings occur in \texttt{WorkerPool::enqueuePreparedSlices}
(commit \texttt{675ca5c6}), where batched enqueue groups slices by owner thread
using \texttt{peerNicPath()} which now returns a reference to the interned string.

\textbf{Generalization} (commit \texttt{12369c0e}): \texttt{TenantIdRef} applies
the same interning pattern to tenant identifiers in the Store, reducing
\texttt{NormalizeTenantIdRef()} from $\Theta(\ell)$ string copy to $\Theta(1)$
pointer comparison.
\end{proof}

\begin{corollary}[Memory Bound]
The intern pool's total memory is bounded by:
\begin{equation}
  M_{\text{pool}} \;\leq\; |\mathcal{P}| \cdot (\ell + C_{\text{hash}} + C_{\text{ptr}})
  \;\approx\; |\mathcal{P}| \cdot (\ell + 64)\ \text{bytes}
\end{equation}
For a cluster with 100 peers $\times$ 8 NICs, $\ell = 40$: $\approx 83$~KB---negligible.
\end{corollary}

\begin{theorem}[Mutex Coalescing Speedup]
\label{thm:mutex-coalesce}
Let $n$ slices be distributed across $w$ worker threads.  Without coalescing,
each slice acquires and releases a mutex independently:
\begin{equation}
  T_{\text{serial}} \;=\; n \cdot (\alpha_{\text{lock}} + \alpha_{\text{enqueue}} +
  \alpha_{\text{unlock}})
\end{equation}
With coalescing, the lock is acquired once per non-empty worker group:
\begin{equation}
  T_{\text{coalesced}} \;=\; w \cdot (\alpha_{\text{lock}} + \alpha_{\text{unlock}})
  \;+\; n \cdot \alpha_{\text{enqueue}}
\end{equation}
The speedup ratio is:
\begin{equation}
  S \;=\; \frac{T_{\text{serial}}}{T_{\text{coalesced}}}
  \;\approx\; \frac{n}{w}
  \quad\text{when } \alpha_{\text{lock}} \gg \alpha_{\text{enqueue}}
  \label{eq:mutex-speedup}
\end{equation}
\end{theorem}

\begin{proof}
The grouping pass is $\Theta(n)$ with no synchronization.
Each \texttt{enqueueSlicesToOwner} call acquires \texttt{worker\_slice\_queue\_lock\_[owner]}
once.  The lock overhead $\alpha_{\text{lock}} \approx 200$--$2000$~ns (futex).
The enqueue $\alpha_{\text{enqueue}} \approx 10$--$50$~ns (\texttt{push\_back}).
For typical $n/w \geq 10$, lock overhead dominates: $S \approx n/w$.
\end{proof}

\begin{lemma}[Atomic Rail Count Fast Path]
\label{lem:rail-count}
The \texttt{paused\_rail\_count\_} atomic allows the hot-path \texttt{isRailAvailable()}
to skip the mutex when no rails are paused:
\begin{equation}
  T_{\text{rail\_check}} \;=\;
  \begin{cases}
    \alpha_{\text{atomic\_load}} \approx 5\,\text{ns} & \text{if no rails paused}\\
    \alpha_{\text{lock}} + \alpha_{\text{scan}} & \text{if any rail paused}
  \end{cases}
\end{equation}
Since rail pauses are rare (error recovery), $\Expect[T_{\text{rail\_check}}] = \Theta(1)$.
\end{lemma}

\begin{theorem}[M/M/1/K Bounded Queue Delay]
\label{thm:mm1k}
Model the TCP transport as an M/M/1/K queue with arrival rate $\lambda$,
service rate $\mu$ per lane, $c$ lanes, and byte capacity $K$.
The traffic intensity is $\rho = \lambda / (c\mu)$.  The blocking probability is:
\begin{equation}
  P_{\text{block}} \;=\;
  \frac{(1-\rho)\,\rho^K}{1-\rho^{K+1}}
  \label{eq:blocking-prob}
\end{equation}
The effective throughput is $\lambda_{\text{eff}} = \lambda(1 - P_{\text{block}})$.
The mean queue length (in bytes) is:
\begin{equation}
  L \;=\; \frac{\rho}{1-\rho} - \frac{(K+1)\rho^{K+1}}{1-\rho^{K+1}}
\end{equation}
\end{theorem}

\begin{proof}
Standard M/M/1/K result from birth-death process with state space $\{0,1,\ldots,K\}$,
birth rate $\lambda$ (when state $< K$), death rate $c\mu$.  The steady-state probabilities
are $\pi_j = \pi_0 \rho^j$ with normalization $\sum_{j=0}^K \pi_j = 1$, giving
$\pi_0 = (1-\rho)/(1-\rho^{K+1})$ for $\rho \neq 1$.  The blocking probability is $\pi_K$.
The implementation's \texttt{queued\_byte\_capacity} is exactly $K$.
\end{proof}

\begin{definition}[EDF Scheduling Problem]
\label{def:edf}
Given $n$ transfer owners, each with data size $w_i$ bytes, deadline $d_i$, and
arrival time $a_i$, the EDF policy dispatches in order of non-decreasing $d_i$,
breaking ties by FIFO (arrival order).
\end{definition}

\begin{theorem}[EDF Optimality for Deadline Minimization]
\label{thm:edf-optimal}
Among all non-preemptive scheduling policies on a single shared resource,
EDF minimizes the maximum lateness $L_{\max} = \max_i (C_i - d_i)$ where $C_i$
is the completion time of owner $i$.  On the TENT admission queue with
\texttt{deadline\_aware = true}:
\begin{equation}
  L_{\max}^{\text{EDF}} \;\leq\; L_{\max}^{\pi}
  \quad\forall\,\text{non-preemptive } \pi
\end{equation}
\end{theorem}

\begin{proof}
Classical result (Jackson 1955, Horn 1974). Consider any schedule $\pi$ and two
adjacent tasks $i, j$ where $d_i > d_j$ but $i$ is scheduled before $j$. Swapping
them does not increase $L_{\max}$: the lateness of $j$ decreases by $p_i$
(processing time of $i$), while the lateness of $i$ increases by $p_j$, but since
$d_i > d_j$, the new lateness of $i$ is still bounded by the old lateness of $j$.
By iterating adjacent swaps, any schedule can be transformed to EDF without
increasing $L_{\max}$.

\textbf{Implementation complexity.} The commit's initial version used
\texttt{std::stable\_sort} in \texttt{pickForDispatch}: $O(N \log N)$ per dispatch call.
The subsequent refinement moves to ordered insertion in \texttt{tryAdmit}:
$O(N)$ per admission (binary search on deque), $O(\text{picked})$ per dispatch.
This amortizes to $O(\log N)$ per owner when the queue maintains sorted invariant.

\textbf{Measured transition band:} On H20/CX-7 RoCEv2, the deadline sweep shows
mean MLU 1.31 at $\sim$200$\mu$s with $\sim$13\% feasible transfers---exactly
where EDF reordering has leverage.
\end{proof}

\begin{lemma}[Undeadlined Owner Deferral]
\label{lem:undeadlined}
Owners with $d_i = 0$ (no deadline) are mapped to $d_i = \texttt{UINT64\_MAX}$ by
\texttt{deadlineKey()}, ensuring they sort after all deadlined owners. This preserves
FIFO ordering among undeadlined owners while never allowing them to preempt
deadline-constrained transfers:
\begin{equation}
  \forall\, i \text{ deadlined}, j \text{ undeadlined}: \;
  \text{dispatch\_order}(i) < \text{dispatch\_order}(j)
\end{equation}
\end{lemma}

\begin{definition}[Degradation Policy]
\label{def:degradation}
A degradation policy $\mathcal{D}$ consists of:
\begin{enumerate}[nosep]
  \item A feasibility threshold $\theta_{\text{local}} > 0$
  \item A bandwidth provider $B(t): \Real^+ \to \Real^+$
  \item A callback \texttt{on\_local\_decode\_suggested} for degraded owners
\end{enumerate}
An owner $i$ with data size $w_i$ and deadline $d_i$ is deemed infeasible when:
\begin{equation}
  \text{MLU}_i(t) \;=\; \frac{(w_i^{\text{ahead}} + w_i) / B(t)}{d_i - t}
  \;\geq\; \theta_{\text{local}}
  \quad\text{or}\quad d_i \leq t
  \label{eq:drop-criterion}
\end{equation}
where $w_i^{\text{ahead}}$ is the bytes queued ahead of owner $i$.
\end{definition}

\begin{theorem}[Degradation Soundness]
\label{thm:degradation}
Under the degradation policy $\mathcal{D}$ with threshold $\theta_{\text{local}}$:
\begin{enumerate}[nosep]
  \item \textbf{No false drops at $\theta_{\text{local}} \geq 1$:}
    Any owner with MLU $\geq 1$ will provably miss its deadline under current
    bandwidth, so dropping it wastes no feasible transfer.
  \item \textbf{Bandwidth reclamation:}
    Dropping infeasible owners frees $w_i$ bytes of NIC capacity, improving
    the MLU of remaining feasible owners by factor
    $\frac{W_{\text{total}}}{W_{\text{total}} - \sum_{j \in \text{dropped}} w_j}$.
  \item \textbf{Local-decode latency:}
    The degradation callback allows the inference engine to recompute the KVCache
    locally rather than waiting for a transfer that would miss its deadline.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) When $\text{MLU}_i \geq 1$, the predicted transfer time $\frac{w_i^{\text{ahead}} + w_i}{B(t)}$
exceeds the remaining window $d_i - t$. Since $B(t)$ is the measured bandwidth
(EWMA, Theorem~\ref{thm:ewma}), the prediction is reliable within estimation error.
(2) The NIC is a shared resource with total capacity $B(t)$. Removing infeasible owners
reduces the numerator $w_i^{\text{ahead}}$ for all subsequent owners.
(3) The callback is raised before the deadline, giving the inference engine time to
compute the KVCache locally.

\textbf{Measured:} At the 100$\mu$s deadline tier, mean MLU is 2.63, well past
$\theta_{\text{local}} = 1.5$, confirming these transfers are provably infeasible.
\end{proof}

\begin{definition}[Exponentially Weighted Moving Average]
\label{def:ewma}
For a time series of bandwidth observations $\{b_1, b_2, \ldots\}$ with smoothing
factor $\alpha \in (0, 1)$:
\begin{equation}
  \hat{B}_t \;=\; \alpha \cdot b_t + (1 - \alpha) \cdot \hat{B}_{t-1}
  \;=\; \sum_{k=0}^{t-1} \alpha(1-\alpha)^k b_{t-k} + (1-\alpha)^t \hat{B}_0
\end{equation}
with $\hat{B}_0 = B_{\text{theoretical}}$ (link rate).
\end{definition}

\begin{theorem}[EWMA Convergence and Bias]
\label{thm:ewma}
The EWMA estimator $\hat{B}_t$ satisfies:
\begin{enumerate}[nosep]
  \item \textbf{Unbiased in steady state:} If $b_t \sim \text{i.i.d.}(\mu_B, \sigma_B^2)$,
    then $\Expect[\hat{B}_t] \to \mu_B$ as $t \to \infty$.
  \item \textbf{Variance:}
    $\text{Var}[\hat{B}_t] \to \frac{\alpha}{2-\alpha} \sigma_B^2$ as $t \to \infty$.
  \item \textbf{Effective window:} The half-life is
    $t_{1/2} = -\ln 2 / \ln(1-\alpha) \approx \ln 2 / \alpha$ for small $\alpha$.
  \item \textbf{Tracking lag:} For a step change $\mu_B \to \mu_B'$, the estimator
    reaches $\epsilon$-accuracy in
    $t_\epsilon = \lceil \ln \epsilon / \ln(1-\alpha) \rceil$ observations.
\end{enumerate}
\end{theorem}

\begin{proof}
Standard properties of EWMA.
(1) $\Expect[\hat{B}_t] = \alpha \sum_{k=0}^{t-1} (1-\alpha)^k \mu_B + (1-\alpha)^t \hat{B}_0
= \mu_B(1-(1-\alpha)^t) + (1-\alpha)^t \hat{B}_0 \to \mu_B$.
(2) $\text{Var}[\hat{B}_t] = \alpha^2 \sum_{k=0}^{t-1} (1-\alpha)^{2k} \sigma_B^2
= \frac{\alpha^2}{1-(1-\alpha)^2} \sigma_B^2 = \frac{\alpha}{2-\alpha} \sigma_B^2$.
(3-4) Direct from geometric decay.

In Mooncake, each NIC maintains two EWMA series:
the \textit{selection EWMA} (for device selection scoring) and the
\textit{transmit estimate} (for deadline prediction). The transmit estimate
uses \texttt{bytes\_ahead}: for the admission queue, this is every RDMA owner already
dispatched and not yet completed; for bandwidth arbitration, it is the NIC's posted
bytes (hardware-side).
\end{proof}

\begin{corollary}[Predicted MLU from Wire Rate]
\label{cor:predicted-mlu}
The predicted MLU at the arbitration point uses the transmit-estimate bandwidth
$\hat{B}_t$ and posted bytes $w_{\text{ahead}}$:
\begin{equation}
  \widehat{\text{MLU}}_i \;=\;
  \frac{(w_{\text{ahead}} + w_i) / \hat{B}_t}{d_i - t}
\end{equation}
This is the formula from \texttt{PredictedMlu()} in \texttt{bw\_arbitration.h:60--66}.
\end{corollary}

\begin{definition}[Machine Load Unit (MLU)]
\label{def:mlu}
For a transfer slice $s$ with data size $|s|$ bytes, estimated NIC bandwidth $B_w$,
and remaining deadline window $\Delta_t = d_s - t_{\text{now}}$:
\begin{equation}
  \text{MLU}(s) \;=\; \frac{|s| / B_w}{\Delta_t}
  \;=\; \frac{|s|}{B_w \cdot \Delta_t}
\end{equation}
MLU $> 1$: will miss deadline. MLU $\to \infty$: past-due. MLU $= 0$: no deadline.
\end{definition}

\begin{theorem}[Optimality of MLU-Priority Scheduling]
\label{thm:mlu-sched}
Given $n$ flows contending for a shared NIC of bandwidth $B_w$, each flow $i$ with
remaining work $w_i$ bytes and deadline $d_i$, the MLU-descending ordering minimizes
the number of deadline violations among all non-preemptive orderings.
When all $w_i$ are equal, this reduces to EDF.
\end{theorem}

\begin{proof}
Exchange argument: consider adjacent flows $i, j$ with $\text{MLU}(i) < \text{MLU}(j)$.
Swapping to put $j$ first:
\begin{align*}
  \text{Finish}(j)_{\text{new}} &= t_0 + w_j/B_w \leq d_j
  \quad\text{(unchanged or improved for $j$)}\\
  \text{Finish}(i)_{\text{new}} &= t_0 + (w_j + w_i)/B_w
\end{align*}
Since $w_j / (B_w(d_j - t_0)) > w_i / (B_w(d_i - t_0))$, the more urgent flow $j$
has a tighter constraint and benefits from earlier scheduling.

\textbf{Measured:} H20/ConnectX-200G-RoCE, 16 threads (4 tight + 12 loose), 1~MB:
tight flows 12.1 $\to$ 44.6~GB/s (+269\%), total throughput unchanged at 48.4~GB/s.
\end{proof}

\begin{definition}[Count-Min Sketch]
\label{def:cms}
A Count-Min Sketch $\text{CMS}(w, d)$ is a $d \times w$ matrix of counters with
$d$ independent hash functions $h_1, \ldots, h_d : \mathcal{U} \to [w]$.
For a stream of elements, \texttt{increment(x)} sets
$T[i][h_i(x)] \mathrel{+}= 1$ for all $i \in [d]$.
The frequency estimate is $\hat{f}(x) = \min_i T[i][h_i(x)]$.
\end{definition}

\begin{theorem}[CMS Error Bound]
\label{thm:cms-error}
For a Count-Min Sketch with width $w$ and depth $d$, after $N$ total increments,
the frequency estimate $\hat{f}(x)$ satisfies:
\begin{equation}
  \hat{f}(x) \;\geq\; f(x) \quad\text{(always, no false negatives)}
\end{equation}
\begin{equation}
  \Prob\bigl[\hat{f}(x) \leq f(x) + \epsilon N\bigr] \;\geq\; 1 - \delta
  \quad\text{with } w = \lceil e/\epsilon \rceil,\; d = \lceil \ln(1/\delta) \rceil
\end{equation}
For Mooncake's parameters $(w=4096, d=4)$:
$\epsilon = e/4096 \approx 0.066\%$ and $\delta = e^{-4} \approx 1.8\%$.
\end{theorem}

\begin{proof}
Standard CMS result. By Markov inequality on a single row, $\Prob[\hat{f}_i(x) - f(x)
\geq \epsilon N] \leq 1/w$ (since the expected collision count is $N/w = \epsilon N$
with $w = e/\epsilon$). Taking the minimum across $d$ independent rows:
$\Prob[\hat{f}(x) - f(x) \geq \epsilon N] \leq (1/w)^d = e^{-d} = \delta$.
\end{proof}

\begin{theorem}[Frequency Admission Policy]
\label{thm:freq-admission}
The local hot cache admits a key $x$ only if its CMS estimate exceeds a threshold
$\theta_f$:
\begin{equation}
  \text{admit}(x) \;=\; \Ind[\hat{f}(x) \geq \theta_f]
\end{equation}
Combined with LRU eviction, this approximates the W-TinyLFU policy
(Einziger \& Friedman, 2014), which achieves near-optimal hit rates under
non-stationary access patterns.

The automatic decay (right-shift all counters every $w \cdot d$ increments) implements
aging with half-life $\sim w \cdot d$ operations, preventing stale frequency estimates
from blocking new hot items.
\end{theorem}

\begin{definition}[Rail State Machine]
\label{def:rail-fsm}
Each RDMA rail $(l, r)$ (local NIC, remote NIC) operates as a finite state machine
$\mathcal{M} = (\{C, O, H\}, \Sigma, \delta, C)$ where:
\begin{itemize}[nosep]
  \item $C$ = Closed (healthy, all transfers admitted)
  \item $O$ = Open (paused, cooldown timer running, periodic probes)
  \item $H$ = Half-Open (cooldown expired, exactly one trial transfer admitted)
  \item $\Sigma = \{\text{fail}, \text{success}, \text{cooldown\_expired}, \text{probe\_interval}\}$
\end{itemize}
\end{definition}

\begin{theorem}[Rail Recovery Convergence]
\label{thm:rail-recovery}
Under the half-open state machine with exponential backoff cooldown
$\tau_k = \min(\tau_0 \cdot 2^k, \tau_{\max})$ where $k$ is the consecutive-failure count:
\begin{enumerate}[nosep]
  \item \textbf{Bounded probe rate:} A permanently failed rail generates at most
    $\frac{1}{\tau_{\text{probe}}}$ probes per second, where
    $\tau_{\text{probe}} = $ \texttt{rail\_probe\_interval\_secs} (default 1s).
  \item \textbf{Recovery latency:} A transiently failed rail (down for time $T_{\text{down}}$)
    recovers within $T_{\text{down}} + \tau_{\text{probe}}$ of becoming healthy
    (probe hits during Open state).
  \item \textbf{No thundering herd:} The Half-Open state admits exactly one trial.
    If it fails, only the cooldown escalates (no flood of slices to a dead peer).
  \item \textbf{Escalation boundedness:}
    \begin{equation}
      \tau_k \;=\; \min\bigl(\tau_0 \cdot 2^k,\; 300\text{s}\bigr)
    \end{equation}
    so the maximum probe spacing is 300 seconds (5 minutes), reached after
    $k = \lceil \log_2(300/\tau_0) \rceil$ consecutive failures.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) In Open state, \texttt{admit()} permits one probe per \texttt{probe\_interval\_}.
The \texttt{probe\_in\_flight} boolean prevents a second probe from racing the first.
(2) If the rail recovers at time $T_{\text{down}}$, the next probe (at most
$\tau_{\text{probe}}$ later) succeeds, triggering \texttt{markRecovered()} which
transitions to Closed.
(3) Half-Open admits exactly one trial via the boolean guard. If the trial fails,
\texttt{markFailed} doubles cooldown and re-enters Open---but does not fire more trials.
(4) Direct from $\tau_k = \min(\tau_0 \cdot 2^k, \tau_{\max})$.

The critical fix over the original (pre-cae02bbc) design is the
\texttt{isAvailable/admit} split: \texttt{isAvailable} is called during
\texttt{updateBestMapping} (the mapping rebuild), and the old single \texttt{available()}
mutated state, causing recursion. The split makes \texttt{isAvailable} a pure predicate.
\end{proof}

\section{Credit-Based Flow Control for Process Groups}
\label{sec:credit-flow}

\begin{commitbox}{84df45a9 --- [PG] Refactor P2PProxy with shared chunk pools and
receiver-driven credit-based flow control}
\textbf{Files:} \texttt{mooncake\_backend.h}, \texttt{p2p\_proxy.cpp},

\begin{definition}[Receiver-Driven Credit Protocol]
\label{def:credit}
In the credit-based flow control for P2P transfers:
\begin{itemize}[nosep]
  \item The receiver pre-allocates $K$ chunk buffers from a shared \texttt{P2PChunkPool}
  \item Each buffer constitutes one \textit{credit}
  \item The sender may transfer up to $K$ chunks in flight; upon completion of each,
    the receiver returns the credit (buffer is recycled)
  \item A per-peer \texttt{generation} counter detects stale credits after failures
\end{itemize}
\end{definition}

\begin{theorem}[Credit Flow Control Throughput]
\label{thm:credit-flow}
With $K$ credits, chunk size $B$, and round-trip latency $\text{RTT}$, the
maximum achievable throughput under credit-based flow control is:
\begin{equation}
  \text{BW}_{\text{max}} \;=\; \frac{K \cdot B}{\text{RTT}}
\end{equation}
The minimum credits to saturate a link of bandwidth $\beta$ is:
\begin{equation}
  K_{\min} \;=\; \left\lceil \frac{\beta \cdot \text{RTT}}{B} \right\rceil
  \;=\; \text{BDP} / B
\end{equation}
where BDP $= \beta \cdot \text{RTT}$ is the bandwidth-delay product.
\end{theorem}

\begin{proof}
Standard sliding-window analysis. At any time, at most $K$ chunks are in flight.
Each chunk takes RTT to complete the round trip (send + receive + credit return).
The pipeline is full when $K \cdot B \geq \beta \cdot \text{RTT}$.

For BI-V100 PCIe P2P ($\beta = 16$~GB/s, RTT $\approx 10\mu$s intra-node,
$B = 1$~MB): $K_{\min} = \lceil 16 \times 10^9 \times 10 \times 10^{-6} / 10^6 \rceil
= \lceil 0.16 \rceil = 1$. For cross-node ($\text{RTT} \approx 100\mu$s):
$K_{\min} = 2$.
\end{proof}

\label{def:pcie-graph}

\begin{itemize}[nosep]
  \item $V = V_{\text{GPU}} \cup V_{\text{NIC}} \cup V_{\text{CPU}} \cup V_{\text{SW}}$
  \item $(u,v) \in E$ if $u$ and $v$ share a PCIe switch or root complex
  \item $w(u,v)$ is the bandwidth (GT/s $\times$ lane width)
\end{itemize}
\end{definition}

\begin{theorem}[Optimal Affinity Assignment]
\label{thm:affinity}
Given $g$ GPUs and $n$ NICs, the optimal GPU$\to$NIC assignment that minimizes total
PCIe hop latency is the solution to the assignment problem:
\begin{equation}
  \min_{\sigma \in S_g} \;\sum_{i=1}^{g} d(G_i, N_{\sigma(i)})
\end{equation}
This is solvable in $\BigO(g^3)$ by the Hungarian algorithm.
\end{theorem}

\begin{corollary}[BI-V100 4-GPU PCIe Topology]
\label{cor:biv100-topo}
On a 4$\times$BI-V100 node with 2 CPU sockets:
\begin{equation}
  d(G_i, G_j) \;=\;
  \begin{cases}
    0 & i = j\\
    1 & \text{same PCIe switch}\\
    2 & \text{same socket, different switch}\\
    3 & \text{cross-socket (QPI/UPI)}
  \end{cases}
\end{equation}
GPU pairs with $d=3$ fall back to the NCCL path due to QPI bandwidth penalty
($\sim$38~GB/s vs 16~GB/s direct PCIe P2P).
\end{corollary}

\begin{theorem}[NUMA-Aware Memory Registration]
\label{thm:numa-mr}
For RDMA memory registration on NUMA architectures (commit \texttt{9e6872fc}),
the post-registration NUMA resolution ensures that staging buffers are allocated
on the same NUMA node as the NIC's PCIe root:
\begin{equation}
  T_{\text{DMA}}(\text{local NUMA}) \;\approx\; \frac{1}{2}\,T_{\text{DMA}}(\text{remote NUMA})
\end{equation}
This is because cross-NUMA DMA traverses the QPI/UPI interconnect, doubling latency.
\end{theorem}

\begin{theorem}[THP Interleaved Allocation]
\label{thm:thp-interleave}
For EFA host segments backed by Transparent Huge Pages (commit \texttt{c3fa13ec}),
interleaving pages across NUMA nodes provides:
\begin{equation}
  \text{BW}_{\text{interleaved}} \;=\; \min\!\left(
    N_{\text{NUMA}} \cdot \text{BW}_{\text{single\_NUMA}},\;
    \text{BW}_{\text{EFA\_link}}
  \right)
\end{equation}
This eliminates NUMA-local memory exhaustion as a bottleneck: the allocation
draws from both nodes equally, and the DMA controller's memory access pattern
(sequential across huge pages) naturally distributes load.
\end{theorem}

\begin{theorem}[Chunked MR Registration]
\label{thm:chunked-mr}

registered as $\lceil S/M \rceil$ separate MRs. The total registration time is:
\begin{equation}

\end{equation}

$B_{\text{pin}}$ is the page-pinning bandwidth.

The batch MR bound (commit \texttt{cfafbfb7}) limits concurrent registrations to
$k_{\text{max}}$, preventing a single large buffer from monopolizing the registration path:
\begin{equation}

  \cdot (k_{\text{max}} \cdot \alpha_{\text{mr}} + M \cdot k_{\text{max}} / B_{\text{pin}})
\end{equation}
\end{theorem}

\begin{commitbox}{3ac7de25 --- [Store] Add lock-free MmapArena allocator for buffer mmap path}
\end{commitbox}

\begin{theorem}[Lock-Free Bump Allocator Throughput]
\label{thm:lock-free-arena}
A lock-free bump allocator using \texttt{compare\_exchange\_weak} on an atomic offset
achieves per-thread allocation throughput:
\begin{equation}
  \text{ops/s} \;\approx\; \frac{f_{\text{CPU}}}{\alpha_{\text{CAS}} + p_{\text{retry}} \cdot \alpha_{\text{CAS}}}
  \;=\; \frac{f_{\text{CPU}}}{\alpha_{\text{CAS}}} \cdot \frac{1}{1 + p_{\text{retry}}}
\end{equation}
where $\alpha_{\text{CAS}} \approx 20$~cycles (uncontended) or $\sim$100~cycles (contended),
and $p_{\text{retry}}$ is the CAS failure probability under $t$ concurrent threads:
\begin{equation}
  p_{\text{retry}} \;\approx\; 1 - \left(1 - \frac{1}{t}\right)^{t-1}
  \;\approx\; 1 - e^{-1} \;\approx\; 0.63 \quad\text{for large } t
\end{equation}
This gives $\sim$2.7$\times$ fewer allocations per second than uncontended, but avoids
the $O(1000)$-cycle mutex overhead entirely. The arena pre-maps a large region via
\texttt{mmap(MAP\_ANONYMOUS | MAP\_HUGETLB)}, so each allocation is a pointer bump
with no syscall.
\end{theorem}

\section{Multi-Tenant Quota and Admission}
\label{sec:tenant-quota}

\begin{commitbox}{98ebebf4 --- [Store] Strict multi-tenant quota admission;
15972687 --- [Store] Make tenant quota charge and release lock-free}
\end{commitbox}

\begin{definition}[Tenant Quota Model]
\label{def:tenant-quota}
Each tenant $t$ has a capacity quota $Q_t$ (bytes). The system maintains:
\begin{itemize}[nosep]
  \item $U_t$: current usage (charged bytes)
  \item Admission predicate: $\text{admit}(t, s) \iff U_t + s \leq Q_t$
  \item Charge: $U_t \mathrel{+}= s$ (atomic)
  \item Release: $U_t \mathrel{-}= s$ (atomic)
\end{itemize}
\end{definition}

\begin{theorem}[Lock-Free Quota Atomicity]
\label{thm:lock-free-quota}
The lock-free quota implementation (commit \texttt{15972687}) uses
\texttt{fetch\_add / fetch\_sub} with relaxed memory order for charge/release,
achieving:
\begin{equation}
  T_{\text{quota\_op}} \;=\; \alpha_{\text{fetch\_add}} \;\approx\; 10\text{--}30\,\text{ns}
\end{equation}
versus the prior mutex-protected implementation:
$T_{\text{quota\_op}}^{\text{mutex}} \approx 200\text{--}2000$~ns.
The speedup is $\sim$10$\times$--$100\times$ on contended paths.

The relaxed ordering is sound because the quota check is a soft admission control:
a transient over-admission of $\Delta = O(\text{threads} \times \text{max\_object\_size})$
is bounded and acceptable (the system evicts to restore capacity).
\end{theorem}

\begin{theorem}[Cache Hit Rate under LRU with Replication]
\label{thm:cache-hit}
Let total capacity be $C$ blocks, working set $W$, replication factor $r$ for hot
blocks (fraction $h$).  Effective unique capacity:
\begin{equation}
  C_{\text{eff}} \;=\; C - h \cdot W \cdot (r - 1)
\end{equation}
LRU hit rate under IRM with Zipf($\alpha$) via Che approximation:
\begin{equation}
  P_{\text{hit}} \;\approx\; 1 - \left(\frac{C_{\text{eff}}}{W}\right)^{1-1/\alpha}
  \quad\text{for } \alpha > 1
\end{equation}
\end{theorem}

\begin{theorem}[Frequency-Gated Admission Optimality]
\label{thm:freq-gated}
Combining the Count-Min Sketch admission gate (Theorem~\ref{thm:freq-admission})
with LRU eviction approximates W-TinyLFU, which satisfies:
\begin{equation}
  P_{\text{hit}}^{\text{TinyLFU}} \;\geq\; P_{\text{hit}}^{\text{LRU}}
  \;\geq\; P_{\text{hit}}^{\text{LFU}(\text{static})}
\end{equation}
under non-stationary workloads, because the CMS decay mechanism (halving every
$w \cdot d$ increments) adapts to distribution shift with time constant $O(w \cdot d)$.
\end{theorem}


%% CHAPTER 21: DISPATCH OVERHEAD DOMINANCE
\section{Dispatch Overhead Dominance on BI-V100}
\label{sec:dispatch}

\begin{commitbox}{infccl/README.md --- Dispatch overhead dominance on BI-V100}
\end{commitbox}

\begin{theorem}[Dispatch vs.\ Data Transfer Crossover]
\label{thm:dispatch-dominance}
For allreduce of $n$ bytes across $p$ GPUs over PCIe with bandwidth $\beta$ and
dispatch latency $\alpha$:
\begin{equation}
  T_{\text{allreduce}}(n) \;=\; \alpha \;+\; \frac{2(p-1)}{p} \cdot \frac{n}{\beta}
\end{equation}
Dispatch dominates when $n < \frac{\alpha \cdot \beta \cdot p}{2(p-1)}$.
For BI-V100 ($\alpha = 50\mu$s, $\beta = 16$~GB/s): $n_{\text{crossover}} \approx 400$~KB.
\end{theorem}

\begin{proposition}[Optimal Threshold Selection]
\label{prop:threshold}
The optimal threshold $n^*$ satisfies $T_{\text{P2P}}(n^*) = T_{\text{NCCL}}(n^*)$:
\begin{equation}
  n^* \;=\; \frac{(\alpha_{\text{NCCL}} - \alpha_{\text{P2P}}) \cdot p \cdot
  \beta_{\text{P2P}} \cdot \beta_{\text{ring}}}
  {p \cdot \beta_{\text{ring}} - 2(p-1) \cdot \beta_{\text{P2P}}}
\end{equation}
For BI-V100: $n^* \approx 5.4$~MB $\to$ threshold set at 8~MB with safety margin.
\end{proposition}

Steps $p$..$2p{-}3$: \texttt{prims.directRecvCopyDirectSend} --- allgather broadcast.
\end{codeline}

\begin{theorem}[Ring AllReduce Optimality]
\label{thm:ring-optimal}
The ring allreduce achieves the bandwidth lower bound:
\begin{equation}
  T_{\text{ring}}(n, p) \;=\; 2(p-1)\alpha \;+\;
  \frac{2(p-1)}{p} \cdot \frac{n}{\beta} \;+\;
  \frac{(p-1)}{p} \cdot n\gamma
\end{equation}
The bandwidth term $\frac{2(p-1)}{p} \cdot \frac{n}{\beta} \to \frac{2n}{\beta}$
as $p \to \infty$ matches the information-theoretic lower bound.
\end{theorem}

\begin{theorem}[Tree AllReduce Latency Advantage]
\label{thm:tree}
The tree (recursive halving-doubling) allreduce:
\begin{equation}
  T_{\text{tree}}(n, p) \;=\; 2\log_2 p \cdot \alpha \;+\;
  2 \cdot \frac{n}{\beta} \;+\; n\gamma
\end{equation}
Latency term $O(\log p)$ vs ring's $O(p)$, but bandwidth term $2n/\beta$ vs
ring's $2(p-1)n/(p\beta) \approx 2n/\beta$: tree is better for small $n$
(latency-bound), ring for large $n$ (bandwidth-bound).
\end{theorem}

\begin{corollary}[infccl Protocol Selection]
For BI-V100 ($p=4$):
$T_{\text{ring}}(n, 4) = 6 \cdot 30\mu\text{s} + 3n/(2 \cdot 16\text{GB/s})$;
$T_{\text{P2P}}(n, 4) = 30\mu\text{s} + n/(16\text{GB/s})$.
At $n = 1$~MB: ring $\approx 274\mu$s, P2P $\approx 93\mu$s.
Custom P2P wins at all sizes for $p=4$ PCIe; 8~MB threshold as safety boundary.
\end{corollary}
